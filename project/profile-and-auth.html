2. Form 表单验证体系与规则封装 ​

表单输入是用户提交数据的核心关口。Flutter 提供了 Form 与 TextFormField 组件，结合 GlobalKey<FormState> 可以在用户点击提交时集中执行校验。

2.1 表单校验规则工具类 (lib/utils/validators.dart) ​

dart
// lib/utils/validators.dart
class FormValidators {
  FormValidators._();

  // 中国大陆 11 位手机号码正则表达式
  static final RegExp _mobileRegExp = RegExp(r'^1[3-9]\d{9}$');

  /// 手机号校验规则
  static String? validateMobile(String? value) {
    if (value == null || value.trim().isEmpty) {
      return '请输入手机号码';
    }
    if (!_mobileRegExp.hasMatch(value.trim())) {
      return '请输入合法的 11 位手机号码';
    }
    return null; // 返回 null 说明校验通过
  }

  /// 密码强度校验规则
  static String? validatePassword(String? value) {
    if (value == null || value.isEmpty) {
      return '请输入登录密码';
    }
    if (value.length < 6) {
      return '密码长度不能少于 6 位';
    }
    if (value.length > 20) {
      return '密码长度不能超过 20 位';
    }
    return null;
  }
}

// lib/pages/auth/login_page.dart
import 'package:flutter/gestures.dart';
import 'package:flutter/material.dart';
import '../../constants/app_colors.dart';
import '../../utils/validators.dart';
import '../../utils/toast_util.dart';
import '../../api/auth_api.dart';

class LoginPage extends StatefulWidget {
  const LoginPage({super.key});

  @override
  State<LoginPage> createState() => _LoginPageState();
}

class _LoginPageState extends State<LoginPage> {
  // 表单全局控制 Key
  final GlobalKey<FormState> _formKey = GlobalKey<FormState>();

  final TextEditingController _mobileController = TextEditingController();
  final TextEditingController _passwordController = TextEditingController();

  // 是否隐藏密码明文
  bool _obscurePassword = true;
  // 是否同意用户隐私协议
  bool _agreeProtocol = false;
  // 是否正在登录网络请求中
  bool _isLoading = false;

  @override
  void dispose() {
    _mobileController.dispose();
    _passwordController.dispose();
    super.dispose();
  }

  // 提交登录逻辑
  Future<void> _handleLogin() async {
    // 1. 回收键盘
    FocusScope.of(context).unfocus();

    // 2. 检查协议勾选
    if (!_agreeProtocol) {
      ToastUtil.error('请先阅读并勾选用户协议与隐私条款');
      return;
    }

    // 3. 执行 Form 表单校验
    final FormState? form = _formKey.currentState;
    if (form == null || !form.validate()) {
      // 校验未通过，输入框下方会自动飘红提示
      return;
    }

    // 4. 发起 API 请求
    setState(() {
      _isLoading = true;
    });

    try {
      final result = await AuthApi.login(
        mobile: _mobileController.text.trim(),
        password: _passwordController.text,
      );

      if (!mounted) return;

      ToastUtil.success('登录成功，欢迎回来：${result['nickname']}');
      Navigator.of(context).pop(); // 返回上一页
    } catch (e) {
      ToastUtil.error(e.toString().replaceAll('Exception: ', ''));
    } finally {
      if (mounted) {
        setState(() {
          _isLoading = false;
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.white,
      appBar: AppBar(
        backgroundColor: Colors.white,
        elevation: 0,
        leading: IconButton(
          icon: const Icon(Icons.close, color: Colors.black87),
          onPressed: () => Navigator.of(context).pop(),
        ),
      ),
      body: SafeArea(
        child: SingleChildScrollView(
          padding: const EdgeInsets.symmetric(horizontal: 28),
          child: Form(
            key: _formKey,
            autovalidateMode: AutovalidateMode.onUserInteraction,
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                const SizedBox(height: 20),
                const Text(
                  '极客商城・欢迎登录',
                  style: TextStyle(
                    fontSize: 24,
                    fontWeight: FontWeight.bold,
                    color: AppColors.textPrimary,
                  ),
                ),
                const SizedBox(height: 8),
                const Text(
                  '发现全球极客好物，畅享品质生活',
                  style: TextStyle(fontSize: 13, color: AppColors.textMuted),
                ),
                const SizedBox(height: 40),

                // 手机号输入框
                TextFormField(
                  controller: _mobileController,
                  keyboardType: TextInputType.phone,
                  maxLength: 11,
                  decoration: InputDecoration(
                    labelText: '手机号码',
                    hintText: '请输入11位手机号码',
                    prefixIcon: const Icon(Icons.phone_android),
                    counterText: '', // 隐藏默认字数计数器
                    border: OutlineInputBorder(borderRadius: BorderRadius.circular(8)),
                    suffixIcon: _mobileController.text.isNotEmpty
                        ? IconButton(
                            icon: const Icon(Icons.clear, size: 18),
                            onPressed: () => setState(() => _mobileController.clear()),
                          )
                        : null,
                  ),
                  validator: FormValidators.validateMobile,
                  onChanged: (_) => setState(() {}),
                ),
                const SizedBox(height: 20),

                // 密码输入框
                TextFormField(
                  controller: _passwordController,
                  obscureText: _obscurePassword,
                  decoration: InputDecoration(
                    labelText: '登录密码',
                    hintText: '请输入6-20位密码',
                    prefixIcon: const Icon(Icons.lock_outline),
                    border: OutlineInputBorder(borderRadius: BorderRadius.circular(8)),
                    suffixIcon: IconButton(
                      icon: Icon(
                        _obscurePassword ? Icons.visibility_off : Icons.visibility,
                        color: Colors.grey,
                      ),
                      onPressed: () {
                        setState(() {
                          _obscurePassword = !_obscurePassword;
                        });
                      },
                    ),
                  ),
                  validator: FormValidators.validatePassword,
                ),
                const SizedBox(height: 16),

                // 用户协议勾选行
                Row(
                  crossAxisAlignment: CrossAxisAlignment.center,
                  children: [
                    SizedBox(
                      width: 24,
                      height: 24,
                      child: Checkbox(
                        value: _agreeProtocol,
                        activeColor: AppColors.primary,

4. 登录 API 服务层与 Mock 契约 (lib/api/auth_api.dart) ​

dart
// lib/api/auth_api.dart
import 'package:dio/dio.dart';

class AuthApi {
  static final Dio _dio = Dio(BaseOptions(
    connectTimeout: const Duration(seconds: 10),
  ));

  /// 手机号密码登录
  static Future<Map<String, dynamic>> login({
    required String mobile,
    required String password,
  }) async {
    try {
      // 真实后端对接接口
      final response = await _dio.post(
        'https://mock.apifox.cn/m1/geekmall/auth/login',
        data: {'mobile': mobile, 'password': password},
      );

      if (response.statusCode == 200 && response.data['code'] == 200) {
        return response.data['data'] as Map<String, dynamic>;
      }
      throw Exception(response.data['message'] ?? '登录失败，请核对信息');
    } catch (e) {
      // 演示 Mock: 手机号只要符合规则即允许登录并颁发 Token
      await Future.delayed(const Duration(milliseconds: 600));
      return {
        'token': 'GEEK_MALL_JWT_TOKEN_${DateTime.now().millisecondsSinceEpoch}',
        'id': '9527',
        'nickname': '极客先锋_${mobile.substring(7)}',
        'avatar': 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=200&q=80',
        'mobile': mobile,
      };
    }
  }
}

5. 高频避坑指南 ​

1. 软键盘弹起导致像素溢出 (Bottom Overflowed)

在输入框获取焦点时，系统软键盘会占据屏幕下半部分，极易触发 A RenderFlex overflowed by xxx pixels 错误。 最佳解决方案：外层必须包裹 SingleChildScrollView，让内容在键盘弹出时具备向上滑动的弹性空间。

2. 软键盘收起时机

点击登录按钮或点击空白处时，应主动调用 FocusScope.of(context).unfocus() 退出键盘输入态，提供干净干练的交互体验。

不过，此时登录成功拿到的 token 和用户资料仅仅停留在当前页面局部变量中，一旦 App 退出重启就会丢失。在下一篇 GetX 全局状态、持久化与通用弹窗 (P97-P99) 中，我们将引入现代化状态管理库 GetX，实现全局跨组件用户状态共享、shared_preferences 本地 Token 存取以及通用二次确认退出弹窗！
